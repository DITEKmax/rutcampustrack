    ```java
    @Service
    @Transactional
    public class OneOffLessonService {
        // Инжектировать: OneOffLessonRepository, ScheduleItemRepository,
        // AcademicGrpcClient, RequestContext, ApplicationEventPublisher

        public OneOffLesson createOneOffLesson(CreateOneOffLessonRequest request) {
            requireHeadmanForGroup(request.groupId());
            // 1. Lookup semester by date: getActiveSemester() и проверить что date в диапазоне
            //    Если date вне active semester — попробовать найти ближайший через SemesterRepository
            //    Для MVP использовать getActiveSemester() как единственный источник (D-23 говорит "lookup по date")
            Long semesterId = getActiveSemester().getId();

            // 2. Conflict check D-09:
            LocalDate date = request.date();
            Short dayOfWeek = (short) date.getDayOfWeek().getValue(); // 1=Пн..7=Вс → ИСПРАВИТЬ: схема 0=Пн..5=Сб
            // Вычислить dayOfWeek: date.getDayOfWeek().getValue() - 1 (0-indexed Monday)
            // Вычислить weekType (ODD/EVEN) по номеру недели от начала семестра
            WeekType weekType = computeWeekType(date, semesterId);
            boolean hasConflict = scheduleItemRepository.existsActiveTemplateSlot(
                request.groupId(), request.lessonNumber(), (short)(date.getDayOfWeek().getValue()-1), weekType, semesterId);
            if (hasConflict) {
                throw new ConflictException("Слот занят активной шаблонной парой на эту дату. Сначала отмените её через POST /api/schedule/lessons/{id}/cancel");
            }

            // 3. Save (UNIQUE violation → DataIntegrityViolationException → 409)
            OneOffLesson oneOff = new OneOffLesson();
            oneOff.setGroupId(request.groupId());
            oneOff.setSubjectId(request.subjectId());
            oneOff.setSemesterId(semesterId);
            oneOff.setDate(date);
            oneOff.setLessonNumber(request.lessonNumber());
            oneOff.setClassroom(request.classroom());
            oneOff.setCreatedBy(requestContext.getUserId());
            return oneOffLessonRepository.save(oneOff);
            // Event публикуется в 60-04 после добавления классов event
        }

        public void deleteOneOffLesson(Long id) {
            OneOffLesson oneOff = oneOffLessonRepository.findById(id)
                .orElseThrow(() -> new ResourceNotFoundException("OneOffLesson", "id", id));
            requireHeadmanForGroup(oneOff.getGroupId());
            oneOffLessonRepository.delete(oneOff);
            // Event lesson.one_off.cancelled публикуется в 60-04
        }
    }
    ```
