```typescript
import { useMemo } from 'react'
import { Link } from 'react-router'
import { ArrowLeft } from '@phosphor-icons/react'
import { useAuth } from '@/features/auth/AuthProvider'
import { useGroupSubjects } from '@/features/headman/shared/headmanApi'
import { SubjectStatsCard } from './SubjectStatsCard'
import { SubjectStatsLoader } from './SubjectStatsLoader'  // inline helper below — see spec

function computeSemesterStart(): string {
  const now = new Date()
  const year = now.getMonth() >= 8 ? now.getFullYear() : now.getFullYear() - 1
  return `${year}-09-01`
}
function todayIso(): string { return new Date().toISOString().slice(0, 10) }

export function StatsPage() {
  const { user } = useAuth()
  const groupId = user?.groupId
  const subjectsQuery = useGroupSubjects()
  const semesterStart = useMemo(computeSemesterStart, [])
  const semesterEnd = useMemo(todayIso, [])

  if (!groupId) return <div className="p-6">Группа не назначена</div>
  if (subjectsQuery.isLoading) {
    return (
      <div className="p-6">
        <Link to="/group" aria-label="Назад"><ArrowLeft size={20} /></Link>
        <h1 className="text-lg font-semibold mb-4">Статистика</h1>
        {Array.from({ length: 3 }).map((_, i) => (
          <div key={i} className="h-[200px] rounded-lg bg-[var(--bg-surface)] animate-pulse mb-4" />
        ))}
      </div>
    )
  }
  const subjects = subjectsQuery.data ?? []
  if (subjects.length === 0) {
    return (
      <div className="p-6">
        <Link to="/group"><ArrowLeft size={20} /></Link>
        <h1 className="text-lg font-semibold mb-4">Статистика</h1>
        <p className="text-sm text-[var(--text-muted)] text-center py-12">Группе не назначены предметы</p>
      </div>
    )
  }
  return (
    <div className="p-6">
      <Link to="/group" aria-label="Назад"><ArrowLeft size={20} /></Link>
      <h1 className="text-lg font-semibold mb-4">Статистика</h1>
      {subjects.map((subject) => (
        <SubjectStatsLoader
          key={subject.id}
          subject={subject}
          groupId={groupId}
          dateFrom={semesterStart}
          dateTo={semesterEnd}
        />
      ))}
    </div>
  )
}
```

Sort order is a concern of the parent list — in this implementation, because each subject loads async, sorting-by-severity is approximated by the order returned from `subjectsQuery.data`. For the test's sorting behavior, derive an intermediate `subjectsWithStats` array inside the parent when all loaders complete — simpler approach: keep rendering in fetched order but pass a `sortKey` prop the parent can assemble after all loaders resolve.

Simpler deterministic approach for testability: collect all `useJournal` results upfront via an orchestrating hook or `Promise.all` pattern. Since TanStack Query returns individual query objects, render all loaders but sort the subjects array before map using precomputed stats when available; during loading render in original order.

Implementation choice: use a `useSubjectStatsList` helper hook (inline, in same file) that:
1. Calls `useQueries` from @tanstack/react-query with one query per subject for the journal data.
2. Calls `useQueries` again for thresholds.

Cell-to-percentage rule: a cell counts as present if `status ∈ {'present', 'excused', 'free_attendance'}`; excluded entirely if `'cancelled'`; counted as absent if `'absent'`.
