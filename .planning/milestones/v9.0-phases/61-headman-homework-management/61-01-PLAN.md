    public void resolveLesson(ResolveLessonRequest req, StreamObserver<LessonResponse> obs) {

        Lesson lesson = lessonRepository
            .findByGroupDateAndLessonNumber(req.getGroupId(), date, req.getLessonNumber())

        obs.onCompleted();
