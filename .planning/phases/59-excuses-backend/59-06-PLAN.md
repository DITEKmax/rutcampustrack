       ```python
       """Handler for student alert events — sends notifications to individual students."""
       import logging
       from aiogram import Bot
       from bot.services.send_queue import SendTask, TelegramSendQueue
       
       logger = logging.getLogger(__name__)
       
       async def handle_student_alert(event: dict, bot: Bot, academic_client, send_queue: TelegramSendQueue, **kwargs) -> None:
           """Notify student about excuse ticket decision."""
           event_type = event.get("event_type")
           payload = event.get("payload", {})
           
           try:
               user_id = payload["user_id"]
               status = payload["status"]
           except KeyError as exc:
               logger.warning("student alert event missing required field: %s", exc)
               return
           
           decision_comment = payload.get("decision_comment") or ""
           
           if status == "approved":
               text = "✅ Ваш запрос на уважительную причину одобрен."
               if decision_comment:
                   text += f"\n\nКомментарий: {decision_comment}"
           elif status == "rejected":
               text = "❌ Ваш запрос на уважительную причину отклонён."
               text += f"\n\nКомментарий: {decision_comment if decision_comment else 'без комментария'}"
           else:
               logger.debug("handle_student_alert called with unexpected status: %s", status)
               return
           
           # Найти telegram_id студента
           # Стратегия: использовать academic_client для поиска по user_id
           # Если academic_client не имеет get_user(user_id) — попробовать get_user_by_id или аналог
           # Если нет ни одного метода — логировать предупреждение и выйти
           try:
               # Попробовать через get_user или get_user_by_id (проверить AcademicGrpcClient)
               student = await academic_client.get_user(user_id)
               if not student or not student.telegram_id:
                   logger.warning("No telegram_id for user_id=%s, cannot notify", user_id)
                   return
               await send_queue.put(
                   SendTask(
                       coroutine_factory=lambda: bot.send_message(chat_id=student.telegram_id, text=text),
                       user_id=user_id,
                       chat_id=student.telegram_id,
                   )
               )
           except Exception as exc:
               logger.warning("Failed to get student info for user_id=%s: %s", user_id, exc)
       ```
       ВАЖНО: Если academic_client не имеет метода get_user(user_id) — изучить academic_client.py и адаптировать код (может быть get_user_by_id или аналог). Не добавлять новые gRPC методы в Python клиент если уже есть подходящий.
