   ```typescript
   // ──────────────────────────────────────────────────────────────
   // Phase 56: Runtime caching for headman GET endpoints (PWA-HEAD-04)
   // Strategy: Stale-While-Revalidate
   // Cache: headman-api-cache-v1 (24h TTL, 100 entries max, GET-only, 200-only)
   // ──────────────────────────────────────────────────────────────

   export function isHeadmanApiRequest(url: URL): boolean {
     if (url.pathname.startsWith('/api/academic/groups/') &&
         (url.pathname.endsWith('/members') ||
          url.pathname.endsWith('/subjects') ||
          url.pathname.endsWith('/teachers'))) {
       return true
     }
     if (url.pathname.startsWith('/api/academic/subjects')) return true
     if (url.pathname.startsWith('/api/academic/thresholds/resolve')) return true
     if (url.pathname.startsWith('/api/attendance/reports/journal')) return true
     if (url.pathname.startsWith('/api/attendance/excuses/pending')) return true
     if (url.pathname.startsWith('/api/attendance/late-checkins/pending')) return true
     return false
   }

   registerRoute(
     ({ url, request }) => request.method === 'GET' && isHeadmanApiRequest(url),
     new StaleWhileRevalidate({
       cacheName: 'headman-api-cache-v1',
       plugins: [
         new CacheableResponsePlugin({ statuses: [200] }),
         new ExpirationPlugin({
           maxAgeSeconds: 86400,      // 24 hours
           maxEntries: 100,
         }),
       ],
     })
   )
   ```

<threat_model>
## Trust Boundaries

| Boundary | Description |
|----------|-------------|
| Network response → SW cache storage | Cached responses persist across sessions and could be served stale to future users on the same device |
| Cached API response → DOM | Stale cached data (up to 24h old) may render in the UI while a fresh fetch runs in the background |

## STRIDE Threat Register

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-----------------|
| T-56-21 | Information Disclosure | Headman data cached under /api/academic and /api/attendance could leak to the next user of a shared device | mitigate | SW cache is per-origin + per-user-agent. On logout, Phase 29 AuthProvider already clears the TanStack query client; an additional `caches.delete('headman-api-cache-v1')` call on logout would fully evict — this hardening is out of scope for this plan but noted for future. Mitigation baseline: cache TTL capped at 24h, so worst case is 24h residual. |
| T-56-22 | Tampering | Attacker with SW dev-tools access replaces cached API responses with forged data | accept | Local device compromise; any user with SW dev access already has full auth token access. Not a meaningful new boundary. |
| T-56-23 | Denial of Service | Unbounded cache growth exhausts device storage | mitigate | ExpirationPlugin caps at 100 entries; 24h TTL auto-evicts stale responses. |
| T-56-24 | Cache poisoning via 4xx/5xx | Caching a 404 or 500 would poison subsequent reads | mitigate | CacheableResponsePlugin restricts storage to status 200 only. 404 (expected from excuses/late-checkin endpoints that may not exist) is not cached, so next request retries the network. |
| T-56-25 | Mutation caching bypass | Non-GET requests accidentally cached | mitigate | registerRoute matcher explicitly checks `request.method === 'GET'` — PUT/POST/PATCH/DELETE fall through to the default network handler. |
</threat_model>
