    1. **HomeworkInlineFormComponent** (standalone):
    ```typescript
    @Component({
      selector: 'app-homework-inline-form',
      standalone: true,
      imports: [CommonModule, ReactiveFormsModule, MatFormFieldModule, MatInputModule, MatButtonModule],
      animations: [trigger('expand', [
        transition(':enter', [style({height: 0, opacity: 0}), animate('200ms ease-out', style({height: '*', opacity: 1}))]),
        transition(':leave', [animate('150ms ease-in', style({height: 0, opacity: 0}))])
      ])],
      templateUrl: './homework-inline-form.component.html'
    })
    export class HomeworkInlineFormComponent {
      @Input({required: true}) subjectId!: number;
      @Input({required: true}) subjectName!: string;   // readonly display (D-11)
      @Input({required: true}) lessonDate!: string;    // ISO
      @Input({required: true}) lessonNumber!: number;
      @Input({required: true}) groupId!: number;
      @Input({required: true}) semesterId!: number;
      @Input() initialHomework: HomeworkResponse | null = null; // edit mode если не null
      @Output() saved = new EventEmitter<HomeworkResponse>();
      @Output() cancelled = new EventEmitter<void>();

      form = new FormGroup({
        title: new FormControl('', [Validators.required, Validators.maxLength(255)]),
        description: new FormControl('', [Validators.maxLength(4000)]),
        link: new FormControl('', [Validators.maxLength(2048)])
      });

      constructor(private api: HeadmanHomeworkApiService) {}

      ngOnInit() {
        if (this.initialHomework) this.form.patchValue(this.initialHomework);
      }

      submit() {
        if (this.form.invalid) return;
        const v = this.form.getRawValue();
        const obs = this.initialHomework
          ? this.api.update(this.initialHomework.id, { title: v.title!, description: v.description, link: v.link })
          : this.api.create({
              title: v.title!, description: v.description, link: v.link,
              subjectId: this.subjectId, groupId: this.groupId, semesterId: this.semesterId,
              lessonDate: this.lessonDate, lessonNumber: this.lessonNumber
            });
        obs.subscribe({ next: (hw) => this.saved.emit(hw), error: (err) => this.handleError(err) });
      }

      cancel() { this.cancelled.emit(); }
      // handleError — показывать RFC 7807 detail под формой
    }
    ```
