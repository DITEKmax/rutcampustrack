**Implementation** in `services/notification-bot/bot/services/send_queue.py`:

```python
import asyncio
import logging
import time
from dataclasses import dataclass
from typing import Any, Awaitable, Callable, Optional

logger = logging.getLogger(__name__)


@dataclass
class SendTask:
    """A deferred Telegram send operation. Per D-06."""
    coroutine_factory: Callable[[], Awaitable[Any]]
    user_id: Optional[int] = None
    chat_id: Optional[int] = None


class TelegramSendQueue:
    """
    Global throttled send queue. Per D-06, D-07, D-08, D-09.

    asyncio.Queue accepts SendTask from any coroutine.
    Single worker coroutine drains queue using token bucket (30 tokens/sec).
    Retry 3x with backoff [1s, 2s, 4s]. On 429: respect Retry-After.
    After 3 failures: log and skip.
    """

    _RATE = 30           # tokens per second (Telegram global limit)
    _MAX_TOKENS = 30     # burst ceiling
    _RETRY_DELAYS = [1, 2, 4]  # D-08: backoff delays

    def __init__(self) -> None:
        self._queue: asyncio.Queue[SendTask] = asyncio.Queue()
        self._tokens: float = self._MAX_TOKENS
        self._last_refill: float = time.monotonic()
        self._worker_task: Optional[asyncio.Task] = None
        self._total_sent: int = 0
        self._total_failed: int = 0

    def start(self) -> None:
        """Start the single worker coroutine. Call once at bot startup. Per D-07."""
        self._worker_task = asyncio.create_task(self._worker())

    async def put(self, task: SendTask) -> None:
        """Enqueue a send task. Callers await this. Per D-06."""
        await self._queue.put(task)

    async def _worker(self) -> None:
        """Drain queue, pace sends via token bucket. Never exits unless cancelled."""
        while True:
            task = await self._queue.get()
            await self._consume_token()
            await self._send_with_retry(task)
            self._queue.task_done()

    async def _consume_token(self) -> None:
        """Block until a token is available (token bucket algorithm)."""
        while True:
            now = time.monotonic()
            elapsed = now - self._last_refill
            self._tokens = min(self._MAX_TOKENS, self._tokens + elapsed * self._RATE)
            self._last_refill = now
            if self._tokens >= 1.0:
                self._tokens -= 1.0
                return
            sleep_time = (1.0 - self._tokens) / self._RATE
            await asyncio.sleep(sleep_time)

    async def _send_with_retry(self, task: SendTask) -> None:
        """Send with retry logic. Per D-08."""
        # Import here to avoid import-time issues in tests
        try:
            from aiogram.exceptions import TelegramRetryAfter as RetryAfterExc
        except ImportError:
            RetryAfterExc = None

        for attempt, delay in enumerate(self._RETRY_DELAYS + [None], start=1):
            try:
                await task.coroutine_factory()
                self._total_sent += 1
                return
            except Exception as e:
                # Check for TelegramRetryAfter (429)
                if RetryAfterExc is not None and isinstance(e, RetryAfterExc):
                    logger.warning("Telegram 429 — retry after %ds", e.retry_after)
                    await asyncio.sleep(e.retry_after)
                    continue
                # Check duck-typed retry_after attribute (for testing)
                if hasattr(e, 'retry_after'):
                    logger.warning("Telegram 429 — retry after %ds", e.retry_after)
                    await asyncio.sleep(e.retry_after)
                    continue

                if delay is None:
                    logger.error(
                        "Send failed after %d attempts user_id=%s chat_id=%s — skipping",
                        len(self._RETRY_DELAYS), task.user_id, task.chat_id,
                    )
                    self._total_failed += 1
                    return
                logger.warning("Send attempt %d failed — retrying in %ds", attempt, delay)
                await asyncio.sleep(delay)

    async def shutdown(self) -> None:
        """Cancel worker and log totals. Per D-09."""
        if self._worker_task:
            self._worker_task.cancel()
            try:
                await self._worker_task
            except asyncio.CancelledError:
                pass
        logger.info("SendQueue shutdown: sent=%d failed=%d", self._total_sent, self._total_failed)
```

Key design decisions enforced:
- D-06: asyncio.Queue + token bucket
- D-07: single global queue, one worker
- D-08: retry 3x [1,2,4], 429 Retry-After, skip after 3
- D-09: counters logged on shutdown

The `_send_with_retry` uses duck-typing for `retry_after` attribute to support both real `TelegramRetryAfter` and mock exceptions in tests. This avoids import-time aiogram dependency in test environment.

Note on retry count: `_RETRY_DELAYS = [1, 2, 4]` plus `[None]` makes 4 iterations. First 3 retry on generic exception with delay, 4th (delay=None) is the final failure — logs and skips. So the factory is called up to 4 times total for generic errors. For 429 errors, the `continue` re-enters the same iteration (does NOT consume a retry slot).
