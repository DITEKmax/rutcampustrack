    1. **StudentHomeworkComponent** (контейнер):
    ```typescript
    @Component({
      selector: 'app-student-homework',
      standalone: true,
      imports: [CommonModule, FormsModule, SegmentedControlComponent,
                StudentHomeworkDayViewComponent, StudentHomeworkWeekViewComponent, StudentHomeworkMonthViewComponent,
                MatSlideToggleModule],
      animations: [routeFade],
      templateUrl: './student-homework.component.html'
    })
    export class StudentHomeworkComponent {
      readonly mode = signal<'day' | 'week' | 'month'>('day');
      readonly selectedDate = signal<Date>(addDays(new Date(), 1)); // tomorrow per D-09
      readonly filterUncompleted = signal(false);
      readonly homeworks = signal<HomeworkItem[]>([]);
      readonly filtered = computed(() =>
        this.filterUncompleted() ? this.homeworks().filter(h => !h.completed) : this.homeworks());

      modeOptions = [
        { value: 'day',   label: 'День' },
        { value: 'week',  label: 'Неделя' },
        { value: 'month', label: 'Месяц' }
      ];

      constructor(private api: StudentApiService) { this.reload(); }

      onDateSelected(d: Date) { this.selectedDate.set(d); this.mode.set('day'); }
      onCompleteToggled(e: { homework: HomeworkItem; completed: boolean }) {
        const req = e.completed ? this.api.markHomeworkComplete(e.homework.id)
                                : this.api.unmarkHomeworkComplete(e.homework.id);
        req.subscribe(() => this.reload());
      }

      private reload() {
        this.api.getHomeworks(groupId, semesterId).subscribe(hw => this.homeworks.set(hw));
      }
    }
    ```
