    1. В `headman-api.service.ts` добавить методы расписания:
    ```typescript
    /** Загрузить шаблон расписания группы для семестра */
    getGroupScheduleItems(groupId: number, semesterId: number): Observable<any> {
      return this.http.get('/api/schedule/schedule-items', {
        params: new HttpParams().set('groupId', groupId).set('semesterId', semesterId),
      });
    }

    /** Создать шаблонный слот расписания (POST /api/schedule/schedule-items) */
    createScheduleItem(body: {
      groupId: number; subjectId: number; semesterId: number;
      dayOfWeek: number; lessonNumber: number; startTime: string;
      endTime: string; weekType: string; room?: string;
    }): Observable<any> {
      return this.http.post('/api/schedule/schedule-items', body);
    }

    /** Обновить шаблонный слот (PUT /api/schedule/schedule-items/{id}) */
    updateScheduleItem(id: number, body: any): Observable<any> {
      return this.http.put(`/api/schedule/schedule-items/${id}`, body);
    }

    /** Удалить (мягко) шаблонный слот (DELETE /api/schedule/schedule-items/{id}) */
    deleteScheduleItem(id: number): Observable<any> {
      return this.http.delete(`/api/schedule/schedule-items/${id}`);
    }

    /** Создать разовую пару */
    createOneOffLesson(body: {
      groupId: number; subjectId: number; date: string;
      lessonNumber: number; classroom?: string;
    }): Observable<any> {
      return this.http.post('/api/schedule/one-off-lessons', body);
    }

    /** Удалить разовую пару */
    deleteOneOffLesson(id: number): Observable<any> {
      return this.http.delete(`/api/schedule/one-off-lessons/${id}`);
    }

    /** Список разовых пар для группы в диапазоне дат */
    getOneOffLessons(groupId: number, dateFrom: string, dateTo: string): Observable<any> {
      return this.http.get('/api/schedule/one-off-lessons', {
        params: new HttpParams().set('groupId', groupId).set('dateFrom', dateFrom).set('dateTo', dateTo),
      });
    }

    /** Отменить конкретную пару по lesson ID (существующий endpoint D-20) */
    cancelLesson(lessonId: number, reason: string): Observable<any> {
      return this.http.post(`/api/schedule/lessons/${lessonId}/cancel`, { reason });
    }
    ```
