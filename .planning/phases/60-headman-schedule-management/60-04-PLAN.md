    ```python
    async def handle_lesson_one_off_created(payload: dict, bot, academic_client) -> None:
        """Уведомляет студентов группы о создании разовой пары."""
        group_id = payload["group_id"]
        date = payload["date"]
        lesson_number = payload["lesson_number"]
        classroom = payload.get("classroom") or "каб. не указан"
        # subject_id для получения названия предмета через academic_client.get_subject(subject_id)
        subject_id = payload["subject_id"]

        students = await academic_client.get_group_students(group_id)  # возвращает list of {telegram_id, ...}
        # D-18: включая старосту — не фильтруем

        text = (
            f"📅 Добавлена разовая пара\n"
            f"Дата: {date}\n"
            f"Пара: {lesson_number}-я\n"
            f"Предмет: {subject_id}\n"  # заменить на название если есть клиент
            f"Кабинет: {classroom}"
        )

        for student in students:
            tg_id = student.get("telegram_id") or student.get("telegramId")
            if tg_id:
                try:
                    await bot.send_message(chat_id=tg_id, text=text)
                except Exception:
                    pass  # студент не подключил бота
    ```
