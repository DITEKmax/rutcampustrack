```typescript
import type { AttendanceStatus } from '@/features/headman/shared/types'

const SEGMENTS: ReadonlyArray<{
  value: AttendanceStatus
  label: string
  ariaLabel: string
}> = [
  { value: 'present',         label: 'б',  ariaLabel: 'Присутствовал' },
  { value: 'absent',          label: 'н',  ariaLabel: 'Отсутствовал' },
  { value: 'excused',         label: 'у',  ariaLabel: 'Уважительная причина' },
  { value: 'free_attendance', label: 'сп', ariaLabel: 'Свободное посещение' },
  { value: 'cancelled',       label: '—',  ariaLabel: 'Отменена' },
]

export interface SegmentedControlProps {
  value: AttendanceStatus
  onValueChange: (status: AttendanceStatus) => void
  disabled?: boolean
  ariaLabel?: string
}

export function SegmentedControl({ value, onValueChange, disabled, ariaLabel }: SegmentedControlProps) {
  return (
    <div
      role="radiogroup"
      aria-label={ariaLabel ?? 'Статус посещения'}
      className="inline-flex gap-1 rounded-md bg-[var(--bg-surface)] p-1"
    >
      {SEGMENTS.map((seg) => {
        const isSelected = seg.value === value
        return (
          <button
            key={seg.value}
            type="button"
            role="radio"
            aria-checked={isSelected}
            aria-label={seg.ariaLabel}
            disabled={disabled}
            onClick={() => !disabled && onValueChange(seg.value)}
            className={[
              'min-w-[40px] min-h-[40px] flex items-center justify-center',
              'text-sm font-semibold rounded-md transition-colors duration-75',
              isSelected
                ? 'bg-[var(--accent-primary)] text-white shadow-sm'
                : 'text-[var(--text-primary)] hover:bg-[var(--bg-elevated)]',
              disabled && 'opacity-50 cursor-not-allowed',
            ].filter(Boolean).join(' ')}
          >
            {seg.label}
          </button>
        )
      })}
    </div>
  )
}
```

```typescript
import { useState } from 'react'
import { AnimatePresence, motion } from 'motion/react'
import { SegmentedControl } from '@/shared/components/SegmentedControl'
import { useMarkAttendance } from '@/features/headman/shared/headmanApi'
import type { AttendanceStatus } from '@/features/headman/shared/types'

export interface JournalStudentRowProps {
  studentId: number
  studentName: string
  lessonId: number
  initialStatus: AttendanceStatus
}

export function JournalStudentRow({ studentId, studentName, lessonId, initialStatus }: JournalStudentRowProps) {
  const [status, setStatus] = useState<AttendanceStatus>(initialStatus)
  const [showError, setShowError] = useState(false)
  const { mutate } = useMarkAttendance()

  const handleChange = (newStatus: AttendanceStatus) => {
    const previous = status
    setStatus(newStatus)        // optimistic
    setShowError(false)
    mutate(
      { lessonId, userId: studentId, status: newStatus },
      {
        onError: () => {
          setStatus(previous)   // revert
          setShowError(true)
          setTimeout(() => setShowError(false), 2000)
        },
      }
    )
  }

  return (
    <div className="relative flex items-center justify-between p-3 rounded-lg bg-[var(--bg-secondary)] mb-2 min-h-[56px]">
      <span className="text-sm font-semibold">{studentName}</span>
      <SegmentedControl
        value={status}
        onValueChange={handleChange}
        ariaLabel={`Посещение — ${studentName}`}
      />
      <AnimatePresence>
        {showError && (
          <motion.div
            initial={{ opacity: 0, y: -4 }}
            animate={{ opacity: 1, y: 0 }}
            exit={{ opacity: 0, y: -4 }}
            transition={{ duration: 0.15 }}
            role="alert"
            className="absolute -top-6 right-0 text-xs font-semibold text-[var(--accent-danger)]"
          >
            Ошибка. Попробуйте ещё раз.
          </motion.div>
        )}
      </AnimatePresence>
    </div>
  )
}
```

## Trust Boundaries

| Boundary | Description |
|----------|-------------|
| Segment tap → PUT mutation | Client-controlled `status` string and path params (lessonId, userId) cross to backend |
| Journal GET response → DOM | Server-provided student names rendered as row text |

## STRIDE Threat Register

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-----------------|
| T-56-13 | Tampering | useMarkAttendance with arbitrary lessonId/userId via dev tools | mitigate | Backend attendance-service validates that: (1) the JWT subject has headman rights over the group containing the lesson, and (2) the userId belongs to that same group. Unauthorized combos → 403. Client row only knows lessonId + studentId values returned by the journal GET for the selected subject/date — no privilege escalation path via UI. |
| T-56-14 | Tampering | status string outside enum (e.g., 'admin') sent in PUT body | mitigate | Backend @Valid + enum deserialization reject unknown values with 400. SegmentedControl restricts client-side to 5 known values. |
| T-56-15 | Information Disclosure | Student names cached in TanStack Query + SW — could leak to other users on shared device | accept | PWA is per-device, auth cleared on logout (Phase 29 AuthProvider already wires queryClient.clear() on logout). SW cache uses the `headman-api-cache-v1` namespace (see Plan 56-06); cache eviction on logout could be added but is out of scope. |
| T-56-16 | XSS | Student name rendered in JournalStudentRow | mitigate | React escapes text children. No `dangerouslySetInnerHTML` — grep gate in success_criteria. |
</threat_model>
